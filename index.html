Código para Google Apps Script

1. Crea una hoja en Google Sheets.
2. Ve a Extensiones > Apps Script.
3. Reemplaza el código con el siguiente fragmento y haz clic en Implementar > Nueva implementación > Aplicación Web (Quién tiene acceso: Cualquier persona). 

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getRange(1, 1, sheet.getLastRow(), sheet.getLastColumn()).getValues();
  return ContentService.createTextOutput(JSON.stringify({status: "ok", data: data}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Fecha", "Nombres", "Cedula", "Provincia", "Ingreso", "Deudas", "Score", "Segmento", "DTI"]);
  }
  
  sheet.appendRow([
    new Date(),
    payload.fullName,
    payload.cedula,
    payload.province,
    payload.netIncome,
    payload.activeDebtsMonthly,
    payload.score,
    payload.segment,
    payload.dti + "%"
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
